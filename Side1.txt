cat > Services/CSideComAdapter.cs << 'EOF'
using System.Runtime.InteropServices;
using Microsoft.Extensions.Logging;
using CSide.ComAdapter.Interfaces;
using CSide.ComAdapter.Models;

namespace CSide.ComAdapter.Services
{
    public class CSideComAdapter : ICSideComAdapter
    {
        private readonly ILogger<CSideComAdapter> _logger;
        private dynamic _csideApp;
        private bool _disposed = false;

        public bool IsConnected { get; private set; }

        public CSideComAdapter(ILogger<CSideComAdapter> logger)
        {
            _logger = logger;
        }

        public Task<bool> ConnectAsync(string server, string database)
        {
            return Task.Run(() =>
            {
                try
                {
                    _logger.LogInformation("Подключение к C/SIDE: {Server}, {Database}", server, database);
                    
                    Type csideType = Type.GetTypeFromProgID("Navision.Application");
                    if (csideType == null)
                    {
                        _logger.LogError("COM тип 'Navision.Application' не найден");
                        return false;
                    }

                    _csideApp = Activator.CreateInstance(csideType);
                    _csideApp.Connect(server, database);
                    
                    IsConnected = true;
                    _logger.LogInformation("Успешно подключено к C/SIDE");
                    return true;
                }
                catch (COMException ex)
                {
                    _logger.LogError(ex, "COM ошибка при подключении к C/SIDE");
                    return false;
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Ошибка при подключении к C/SIDE");
                    return false;
                }
            });
        }

        public Task DisconnectAsync()
        {
            return Task.Run(() =>
            {
                try
                {
                    if (IsConnected && _csideApp != null)
                    {
                        _csideApp.Disconnect();
                        _csideApp = null;
                        IsConnected = false;
                        _logger.LogInformation("Отключено от C/SIDE");
                    }
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Ошибка при отключении от C/SIDE");
                }
            });
        }

        public Task<bool> ExportObjectsAsync(string filePath, int[] objectIds = null)
        {
            if (!IsConnected) throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    dynamic objectManager = _csideApp.ActiveSession.Database.Objects;
                    
                    if (objectIds != null && objectIds.Length > 0)
                    {
                        foreach (int objectId in objectIds)
                        {
                            objectManager.Export(objectId, filePath, true);
                        }
                    }
                    else
                    {
                        objectManager.Export(0, filePath, false);
                    }
                    
                    _logger.LogInformation("C/SIDE объекты экспортированы в {FilePath}", filePath);
                    return true;
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Ошибка экспорта C/SIDE объектов");
                    return false;
                }
            });
        }

        public Task<bool> ImportObjectsAsync(string filePath)
        {
            if (!IsConnected) throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    dynamic objectManager = _csideApp.ActiveSession.Database.Objects;
                    int importedCount = objectManager.Import(filePath);
                    
                    _logger.LogInformation("Импортировано C/SIDE объектов: {Count}", importedCount);
                    return importedCount > 0;
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Ошибка импорта C/SIDE объектов");
                    return false;
                }
            });
        }

        public Task<bool> CompileObjectAsync(int objectType, int objectId)
        {
            if (!IsConnected) throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    dynamic objectManager = _csideApp.ActiveSession.Database.Objects;
                    bool success = objectManager.Compile(objectType, objectId);
                    
                    if (success)
                        _logger.LogInformation("C/SIDE объект {Type}/{Id} скомпилирован", objectType, objectId);
                    else
                        _logger.LogWarning("Ошибка компиляции C/SIDE объекта {Type}/{Id}", objectType, objectId);
                    
                    return success;
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Ошибка компиляции C/SIDE объекта");
                    return false;
                }
            });
        }

        public Task<bool> CompileAllObjectsAsync()
        {
            if (!IsConnected) throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    dynamic objectManager = _csideApp.ActiveSession.Database.Objects;
                    objectManager.CompileAll();
                    
                    _logger.LogInformation("Все C/SIDE объекты скомпилированы");
                    return true;
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Ошибка компиляции всех C/SIDE объектов");
                    return false;
                }
            });
        }

        public Task<dynamic> GetRecordAsync(string tableName, string filterField, object filterValue)
        {
            if (!IsConnected) throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    dynamic record = _csideApp.ActiveSession.Database.DataSet(tableName);
                    record.SetFilter(filterField, filterValue.ToString());
                    
                    if (record.FindFirst())
                        return record;
                    
                    return null;
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Ошибка получения записи из {TableName}", tableName);
                    return null;
                }
            });
        }

        public void Dispose()
        {
            if (!_disposed)
            {
                DisconnectAsync().Wait(5000);
                if (_csideApp != null)
                {
                    Marshal.ReleaseComObject(_csideApp);
                    _csideApp = null;
                }
                _disposed = true;
            }
            GC.SuppressFinalize(this);
        }
    }
}

Создаем пример использования:
bash
cat > Examples/DemoExample.cs << 'EOF'
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Logging;
using CSide.ComAdapter.Interfaces;
using CSide.ComAdapter.Services;

namespace CSide.ComAdapter.Examples
{
    public class DemoExample
    {
        public static async Task RunDemo()
        {
            // Настройка DI
            var services = new ServiceCollection();
            services.AddLogging(builder => builder.AddConsole());
            services.AddSingleton<ICSideComAdapter, CSideComAdapter>();
            
            var serviceProvider = services.BuildServiceProvider();
            var logger = serviceProvider.GetRequiredService<ILogger<DemoExample>>();
            var csideAdapter = serviceProvider.GetRequiredService<ICSideComAdapter>();

            try
            {
                // Подключение к C/SIDE
                bool connected = await csideAdapter.ConnectAsync("localhost", "CSIDE_DEMO");
                if (!connected)
                {
                    logger.LogError("Не удалось подключиться к C/SIDE");
                    return;
                }

                // Экспорт C/SIDE объектов
                await csideAdapter.ExportObjectsAsync(@"C:\cside_export.txt");
                
                // Компиляция C/SIDE объектов
                await csideAdapter.CompileAllObjectsAsync();
                
                logger.LogInformation("C/SIDE демо завершено успешно");
            }
            catch (Exception ex)
            {
                logger.LogError(ex, "Ошибка в C/SIDE демо");
            }
            finally
            {
                await csideAdapter.DisconnectAsync();
            }
        }
    }
}

        public Task<IEnumerable<dynamic>> GetRecordsAsync(string tableName, string filter = "")
        {
            if (!IsConnected || _csideApp == null) 
                throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    var records = new List<dynamic>();
                    dynamic record = _csideApp.ActiveSession.Database.DataSet(tableName);
                    
                    if (!string.IsNullOrEmpty(filter))
                    {
                        record.SetFilter(filter);
                    }
                    
                    if (record.FindSet())
                    {
                        do
                        {
                            records.Add(record);
                        } 
                        while (record.FindNext());
                    }
                    
                    _logger.LogInformation("Получено {Count} записей из {TableName}", records.Count, tableName);
                    return records.AsEnumerable();
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Ошибка получения записей из {TableName}", tableName);
                    return Enumerable.Empty<dynamic>();
                }
            });
        }

using System;
using System.Collections.Generic;
using System.Linq;
using System.Runtime.InteropServices;
using System.Threading.Tasks;
using Microsoft.Extensions.Logging;
using CSide.ComAdapter.Interfaces;
using CSide.ComAdapter.Models;

namespace CSide.ComAdapter.Services
{
    public class CSideComAdapter : ICSideComAdapter
    {
        private readonly ILogger<CSideComAdapter> _logger;
        private dynamic _csideApp;
        private bool _disposed = false;

        public bool IsConnected { get; private set; }

        public CSideComAdapter(ILogger<CSideComAdapter> logger)
        {
            _logger = logger;
        }

        public Task<bool> ConnectAsync(string server, string database)
        {
            return Task.Run(() =>
            {
                try
                {
                    LogInformation("Подключение к C/SIDE: {0}, {1}", server, database);
                    
                    Type csideType = Type.GetTypeFromProgID("Navision.Application");
                    if (csideType == null)
                    {
                        LogError("COM тип 'Navision.Application' не найден");
                        return false;
                    }

                    _csideApp = Activator.CreateInstance(csideType);
                    _csideApp.Connect(server, database);
                    
                    IsConnected = true;
                    LogInformation("Успешно подключено к C/SIDE");
                    return true;
                }
                catch (COMException ex)
                {
                    LogError("COM ошибка при подключении к C/SIDE: {0}", ex.Message);
                    return false;
                }
                catch (Exception ex)
                {
                    LogError("Ошибка при подключении к C/SIDE: {0}", ex.Message);
                    return false;
                }
            });
        }

        public Task DisconnectAsync()
        {
            return Task.Run(() =>
            {
                try
                {
                    if (IsConnected && _csideApp != null)
                    {
                        _csideApp.Disconnect();
                        _csideApp = null;
                        IsConnected = false;
                        LogInformation("Отключено от C/SIDE");
                    }
                }
                catch (Exception ex)
                {
                    LogError("Ошибка при отключении от C/SIDE: {0}", ex.Message);
                }
            });
        }

        public Task<bool> ExportObjectsAsync(string filePath, int[] objectIds = null)
        {
            if (!IsConnected || _csideApp == null) 
                throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    dynamic objectManager = _csideApp.ActiveSession.Database.Objects;
                    
                    if (objectIds != null && objectIds.Length > 0)
                    {
                        foreach (int objectId in objectIds)
                        {
                            objectManager.Export(objectId, filePath, true);
                        }
                    }
                    else
                    {
                        objectManager.Export(0, filePath, false);
                    }
                    
                    LogInformation("C/SIDE объекты экспортированы в {0}", filePath);
                    return true;
                }
                catch (Exception ex)
                {
                    LogError("Ошибка экспорта C/SIDE объектов: {0}", ex.Message);
                    return false;
                }
            });
        }

        public Task<bool> ImportObjectsAsync(string filePath)
        {
            if (!IsConnected || _csideApp == null) 
                throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    dynamic objectManager = _csideApp.ActiveSession.Database.Objects;
                    int importedCount = objectManager.Import(filePath);
                    
                    LogInformation("Импортировано C/SIDE объектов: {0}", importedCount);
                    return importedCount > 0;
                }
                catch (Exception ex)
                {
                    LogError("Ошибка импорта C/SIDE объектов: {0}", ex.Message);
                    return false;
                }
            });
        }

        public Task<bool> CompileObjectAsync(int objectType, int objectId)
        {
            if (!IsConnected || _csideApp == null) 
                throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    dynamic objectManager = _csideApp.ActiveSession.Database.Objects;
                    bool success = objectManager.Compile(objectType, objectId);
                    
                    if (success)
                        LogInformation("C/SIDE объект {0}/{1} скомпилирован", objectType, objectId);
                    else
                        LogWarning("Ошибка компиляции C/SIDE объекта {0}/{1}", objectType, objectId);
                    
                    return success;
                }
                catch (Exception ex)
                {
                    LogError("Ошибка компиляции C/SIDE объекта: {0}", ex.Message);
                    return false;
                }
            });
        }

        public Task<bool> CompileAllObjectsAsync()
        {
            if (!IsConnected || _csideApp == null) 
                throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    dynamic objectManager = _csideApp.ActiveSession.Database.Objects;
                    objectManager.CompileAll();
                    
                    LogInformation("Все C/SIDE объекты скомпилированы");
                    return true;
                }
                catch (Exception ex)
                {
                    LogError("Ошибка компиляции всех C/SIDE объектов: {0}", ex.Message);
                    return false;
                }
            });
        }

        public Task<dynamic> GetRecordAsync(string tableName, string filterField, object filterValue)
        {
            if (!IsConnected || _csideApp == null) 
                throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    dynamic record = _csideApp.ActiveSession.Database.DataSet(tableName);
                    record.SetFilter(filterField, filterValue.ToString());
                    
                    if (record.FindFirst())
                        return record;
                    
                    return null;
                }
                catch (Exception ex)
                {
                    LogError("Ошибка получения записи из {0}: {1}", tableName, ex.Message);
                    return null;
                }
            });
        }

        public Task<IEnumerable<dynamic>> GetRecordsAsync(string tableName, string filter = "")
        {
            if (!IsConnected || _csideApp == null) 
                throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    var records = new List<dynamic>();
                    dynamic record = _csideApp.ActiveSession.Database.DataSet(tableName);
                    
                    if (!string.IsNullOrEmpty(filter))
                    {
                        record.SetFilter(filter);
                    }
                    
                    if (record.FindSet())
                    {
                        do
                        {
                            records.Add(record);
                        } 
                        while (record.FindNext());
                    }
                    
                    LogInformation("Получено {0} записей из {1}", records.Count, tableName);
                    return records.AsEnumerable();
                }
                catch (Exception ex)
                {
                    LogError("Ошибка получения записей из {0}: {1}", tableName, ex.Message);
                    return Enumerable.Empty<dynamic>();
                }
            });
        }

        public Task<bool> InsertRecordAsync(string tableName, dynamic record)
        {
            if (!IsConnected || _csideApp == null) 
                throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    record.Insert();
                    LogInformation("Запись добавлена в {0}", tableName);
                    return true;
                }
                catch (Exception ex)
                {
                    LogError("Ошибка добавления записи в {0}: {1}", tableName, ex.Message);
                    return false;
                }
            });
        }

        public Task<bool> ModifyRecordAsync(dynamic record)
        {
            if (!IsConnected || _csideApp == null) 
                throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {

                catch (Exception ex)
                {
                    LogError("Ошибка изменения записи: {0}", ex.Message);
                    return false;
                }
            });
        }

        public Task<bool> DeleteRecordAsync(dynamic record)
        {
            if (!IsConnected || _csideApp == null) 
                throw new InvalidOperationException("Не подключено к C/SIDE");

            return Task.Run(() =>
            {
                try
                {
                    record.Delete();
                    LogInformation("Запись удалена");
                    return true;
                }
                catch (Exception ex)
                {
                    LogError("Ошибка удаления записи: {0}", ex.Message);
                    return false;
                }
            });
        }

        // Вспомогательные методы для логирования
        private void LogInformation(string message, params object[] args)
        {
            if (_logger != null)
                _logger.LogInformation(message, args);
            else
                Console.WriteLine("[INFO] " + string.Format(message, args));
        }

        private void LogWarning(string message, params object[] args)
        {
            if (_logger != null)
                _logger.LogWarning(message, args);
            else
                Console.WriteLine("[WARN] " + string.Format(message, args));
        }

        private void LogError(string message, params object[] args)
        {
            if (_logger != null)
                _logger.LogError(message, args);
            else
                Console.WriteLine("[ERROR] " + string.Format(message, args));
        }

        private void LogError(Exception ex, string message, params object[] args)
        {
            if (_logger != null)
                _logger.LogError(ex, message, args);
            else
                Console.WriteLine("[ERROR] " + string.Format(message, args) + " | Exception: " + ex.Message);
        }

        public void Dispose()
        {
            if (!_disposed)
            {
                try
                {
                    DisconnectAsync().Wait(5000);
                }
                catch { }

                if (_csideApp != null)
                {
                    try
                    {
                        Marshal.ReleaseComObject(_csideApp);
                    }
                    catch { }
                    _csideApp = null;
                }
                _disposed = true;
            }
        }
    }

            // Упрощенная версия без сложных зависимостей
            var csideAdapter = new CSideComAdapter(null);

            try
            {
                // Подключение к C/SIDE
                Console.WriteLine("Подключение к C/SIDE...");
                bool connected = await csideAdapter.ConnectAsync("localhost", "CSIDE_DEMO");
                if (!connected)
                {
                    Console.WriteLine("Не удалось подключиться к C/SIDE");
                    return;
                }

                Console.WriteLine("Успешно подключено к C/SIDE");

                // Экспорт C/SIDE объектов
                Console.WriteLine("Экспорт объектов...");
                await csideAdapter.ExportObjectsAsync(@"C:\cside_export.txt");
                Console.WriteLine("Экспорт объектов завершен");
                
                // Компиляция C/SIDE объектов
                Console.WriteLine("Компиляция объектов...");
                await csideAdapter.CompileAllObjectsAsync();
                Console.WriteLine("Компиляция объектов завершена");
                
                Console.WriteLine("C/SIDE демо завершено успешно");
            }
            catch (Exception ex)
            {
                Console.WriteLine($"Ошибка в C/SIDE демо: {ex.Message}");
                Console.WriteLine($"StackTrace: {ex.StackTrace}");
            }
            finally
            {
                await csideAdapter.DisconnectAsync();
                Console.WriteLine("Отключено от C/SIDE");
                csideAdapter.Dispose();
            }
        }
    }
}
